#!/usr/bin/env python3
"""
Convert bank statement CSV to JSON format for los monitoring ingest.
Expected CSV columns: date, description, amount, currency, balance (typical bank statement)
Output JSON format for los monitoring ingest:
{
  "transactions": [
    {
      "date": "YYYY-MM-DD",
      "description": "string",
      "amount": number,
      "currency": "GBP",
      "balance": number,
      "type": "credit|debit|transfer|fee"
    }
  ]
}
"""

import csv
import json
import sys
from datetime import datetime

def csv_to_json(csv_path, json_path=None):
    transactions = []
    
    with open(csv_path, 'r') as f:
        reader = csv.DictReader(f)
        for row in reader:
            # Normalize column names (case-insensitive)
            row_lower = {k.lower().strip(): v for k, v in row.items()}
            
            # Parse date (try common formats)
            date_str = row_lower.get('date', row_lower.get('transaction_date', ''))
            try:
                # Try DD/MM/YYYY
                dt = datetime.strptime(date_str, '%d/%m/%Y')
                date_iso = dt.strftime('%Y-%m-%d')
            except ValueError:
                try:
                    # Try MM/DD/YYYY
                    dt = datetime.strptime(date_str, '%m/%d/%Y')
                    date_iso = dt.strftime('%Y-%m-%d')
                except ValueError:
                    date_iso = date_str  # keep as-is if already ISO
            
            # Parse amount (handle negative for debits)
            amount_str = row_lower.get('amount', row_lower.get('value', '0')).replace(',', '')
            amount = float(amount_str)
            
            # Determine transaction type
            description = row_lower.get('description', row_lower.get('narration', '')).lower()
            if 'transfer' in description or 'fpo' in description or 'fp' in description:
                tx_type = 'transfer'
            elif amount > 0:
                tx_type = 'credit'
            else:
                tx_type = 'debit'
            
            # Get currency (default GBP for UK loans)
            currency = row_lower.get('currency', 'GBP').upper()
            
            # Get running balance if available
            balance_str = row_lower.get('balance', row_lower.get('closing_balance', ''))
            balance = float(balance_str.replace(',', '')) if balance_str else None
            
            transaction = {
                "date": date_iso,
                "description": row_lower.get('description', row_lower.get('narration', '')),
                "amount": abs(amount),
                "currency": currency,
                "type": tx_type
            }
            if balance is not None:
                transaction["balance"] = balance
            
            transactions.append(transaction)
    
    output = {"transactions": transactions}
    
    if json_path:
        with open(json_path, 'w') as f:
            json.dump(output, f, indent=2)
        print(f"Converted {len(transactions)} transactions to {json_path}")
    else:
        print(json.dumps(output, indent=2))
    
    return output

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Usage: csv_to_json.py <input.csv> [output.json]")
        sys.exit(1)
    
    csv_path = sys.argv[1]
    json_path = sys.argv[2] if len(sys.argv) > 2 else None
    csv_to_json(csv_path, json_path)

#!/bin/bash

# CP Tracking Management Script for Open LOS
# Usage: ./cp_manager.sh <dealId> [command]

DEAL_ID="$1"
COMMAND="$2"

# Colors for output
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# Get deal custom fields
get_deal_custom_fields() {
    los deal get "$DEAL_ID" --format json | jq -r '.custom_fields // {}'
}

# Get deal documents
get_deal_documents() {
    los doc list "$DEAL_ID" --format json | jq -r '.documents // []'
}

# Update deal custom fields
update_deal_custom_fields() {
    local json="$1"
    los deal update "$DEAL_ID" --custom "$json" > /dev/null 2>&1
}

# Sync CP status based on uploaded documents
sync_cp_status() {
    echo "Syncing CP status with uploaded documents..."
    
    local custom_fields
    custom_fields=$(get_deal_custom_fields)
    
    # Check if CP tracking exists
    if [[ $(echo "$custom_fields" | jq -r '.conditions_precedent // empty') == "null" ]]; then
        echo -e "${RED}ERROR: No CP tracking structure found in deal custom fields${NC}"
        echo "Please initialize CP tracking first."
        return 1
    fi
    
    local documents
    documents=$(get_deal_documents)
    
    # Build document name list for matching
    local doc_names
    doc_names=$(echo "$documents" | jq -r '.[].filename')
    
    # Update CP status based on documents
    local updated_cps
    updated_cps=$(echo "$custom_fields" | jq --argjson docs "$documents" '
        .conditions_precedent.checklist |= map(
            .documents = ($docs | map(.filename) | map(select(test("(?i)" + (.required_document_types | join("|"))))) | unique)
            | if (.documents | length) > 0 and .status == "OUTSTANDING" 
              then .status = "RECEIVED" 
              else . 
              end
        )
    ')
    
    update_deal_custom_fields "$updated_cps"
    echo -e "${GREEN}CP status synced successfully${NC}"
}

# Show CP checklist with completion percentage
show_checklist() {
    local custom_fields
    custom_fields=$(get_deal_custom_fields)
    
    if [[ $(echo "$custom_fields" | jq -r '.conditions_precedent // empty') == "null" ]]; then
        echo -e "${RED}ERROR: No CP tracking structure found${NC}"
        return 1
    fi
    
    local total_cps
    total_cps=$(echo "$custom_fields" | jq '.conditions_precedent.checklist | length')
    
    local received_cps
    received_cps=$(echo "$custom_fields" | jq '[.conditions_precedent.checklist[] | select(.status == "RECEIVED")] | length')
    
    local completion_pct
    completion_pct=$((received_cps * 100 / total_cps))
    
    echo "================================================"
    echo "Conditions Precedent Checklist"
    echo "Deal: $DEAL_ID"
    echo "Completion: $received_cps/$total_cps ($completion_pct%)"
    echo "================================================"
    echo ""
    
    echo "$custom_fields" | jq -r '.conditions_precedent.checklist[] | 
        "\(.id) | \(.description) | \(.status) | Documents: \(.documents | join(", "))"' | \
        while IFS='|' read -r id desc status docs; do
            case $status in
                "RECEIVED") color=$GREEN ;;
                "OUTSTANDING") color=$RED ;;
                *) color=$YELLOW ;;
            esac
            printf "${color}%-10s %-50s %-15s %s${NC}\n" "$id" "$desc" "$status" "$docs"
        done
}

# Validate if all CPs are satisfied (for blocking advancement)
validate_completion() {
    local custom_fields
    custom_fields=$(get_deal_custom_fields)
    
    if [[ $(echo "$custom_fields" | jq -r '.conditions_precedent // empty') == "null" ]]; then
        echo -e "${RED}ERROR: No CP tracking structure found${NC}"
        return 1
    fi
    
    local outstanding_count
    outstanding_count=$(echo "$custom_fields" | jq '[.conditions_precedent.checklist[] | select(.status == "OUTSTANDING")] | length')
    
    if [[ $outstanding_count -eq 0 ]]; then
        echo -e "${GREEN}✓ All conditions precedent satisfied. Deal can advance.${NC}"
        return 0
    else
        echo -e "${RED}✗ $outstanding_count condition(s) precedent still outstanding:${NC}"
        echo "$custom_fields" | jq -r '.conditions_precedent.checklist[] | select(.status == "OUTSTANDING") | "  - \(.description)"'
        return 1
    fi
}

# Mark a specific CP as received (with document reference)
mark_cp_received() {
    local cp_id="$2"
    local doc_ref="$3"
    
    local custom_fields
    custom_fields=$(get_deal_custom_fields)
    
    if [[ $(echo "$custom_fields" | jq -r '.conditions_precedent // empty') == "null" ]]; then
        echo -e "${RED}ERROR: No CP tracking structure found${NC}"
        return 1
    fi
    
    # Check if CP exists
    local cp_exists
    cp_exists=$(echo "$custom_fields" | jq --arg id "$cp_id" '[.conditions_precedent.checklist[] | select(.id == $id)] | length')
    
    if [[ $cp_exists -eq 0 ]]; then
        echo -e "${RED}ERROR: CP ID '$cp_id' not found${NC}"
        return 1
    fi
    
    # Update CP status and add document reference
    local updated_cps
    updated_cps=$(echo "$custom_fields" | jq --arg id "$cp_id" --arg doc "$doc_ref" '
        .conditions_precedent.checklist |= map(
            if .id == $id then 
                .status = "RECEIVED" 
                | if $doc != "" and (.documents | index($doc) | not) then 
                    .documents += [$doc] 
                  else 
                    . 
                  end
            else 
                . 
            end
        )
    ')
    
    update_deal_custom_fields "$updated_cps"
    echo -e "${GREEN}CP '$cp_id' marked as RECEIVED${NC}"
}

# Main command dispatcher
case "$COMMAND" in
    "sync")
        sync_cp_status
        ;;
    "list"|"show")

            echo "Usage: $0 <dealId> mark <cp_id> [document_reference]"
            exit 1
        fi
        mark_cp_received "$DEAL_ID" "$3" "$4"
        ;;
    *)
        echo "Usage: $0 <dealId> {sync|list|show|validate|mark <cp_id> [doc_ref]}"
        echo ""
        echo "Commands:"
        echo "  sync              - Sync CP status with uploaded documents"
        echo "  list|show         - Display CP checklist with completion %"
        echo "  validate          - Check if all CPs are satisfied (exit 0 if yes)"
        echo "  mark <cp_id> [doc] - Mark specific CP as received (optional doc reference)"
        exit 1
        ;;
esac
